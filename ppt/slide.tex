\subsection{Problématique}
\begin{frame}
\frametitle{Le problème du \emph{sparse reward}}

Ce phénomène porte un nom : le \textbf{sparse reward} \nocite{narvekarCurriculumLearningReinforcement2020}

\vspace{0.3cm}

\pause

\begin{itemize}
    \item Objectif atteint rarement (voire jamais au début) $\rightarrow$
    presque aucun signal exploitable
    \item Résultat : aucun apprentissage observable, même après des
    millions de pas d'entraînement
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Problématique}



\begin{beamerboxesrounded}[scheme=madrid]{Problématique}
Comment mener l'apprentissage par renforcement dans un environnement
à récompense rare, de la sélection d'algorithmes standards jusqu'à la
conception d'un signal de récompense pour un environnement avancé ?    
\end{beamerboxesrounded}


\vspace{0.5cm}

\pause

\begin{enumerate}
    \item \textbf{Sélection et optimisation des modèles} : quel
    algorithme, quels hyperparamètres ?
    \item \textbf{Construction du signal de récompense} : comment
    densifier un signal rare sans dénaturer la tâche ?
\end{enumerate}

\end{frame}

\begin{frame}
\frametitle{Démarche en deux phases}

\begin{enumerate}
    \item \textbf{Phase I} — Validation méthodologique sur
    \emph{LunarLander} : DQN / PPO / SAC, Optuna, multi-seeds. \pause
    \item \textbf{Phase II} — Retour à \textbf{Rocket League} :
    reward shaping et curriculum learning.
\end{enumerate}

\end{frame}

\section{Phase II}

\subsection{L'environnement Rocket League}
\begin{frame}
\frametitle{L'environnement Rocket League}

\begin{itemize}
    \item \textbf{RocketSim} + \textbf{RLGym} : physique fidèle,
    entraînement \emph{headless} \nocite{ZealanLRocketSim2022,RLGymRLGym2020}
    \item \texttt{rlgym-ppo} et \texttt{rlgym-sac} : forks vectorisés,
    conçus pour l'entraînement à grande échelle (milliards de steps)
    \nocite{AllenRlgymppo2023,McSwainRlgymsac2026}
    \item Pas d'équivalent vectorisé pour DQN $\rightarrow$ écarté pour
    la Phase II
    \item Mode \textbf{1v1}, \emph{self-play}
    \item Espace d'actions réduit : $1944 \rightarrow \textbf{90}$
    actions (PPO)
    \item Décision ramenée de 120~Hz à \textbf{15~Hz}
\end{itemize}

\end{frame}

\subsection{Reward shaping et curriculum}
\begin{frame}
\frametitle{Reward shaping et curriculum}

    \item \textbf{Curriculum learning} : introduction progressive, sans
    dégrader les acquis
\end{itemize}

\end{frame}

\subsection{Reward hacking}
\begin{frame}
\frametitle{Reward hacking}

\begin{beamerboxesrounded}[scheme=madrid]{Contexte}

$R = \text{toucher la balle} + v_{\text{balle} \rightarrow \text{but adverse}}$

\footnotesize (Récompense individuelle)
\vspace{0.1cm}
\end{beamerboxesrounded}

    }

\end{frame}

\begin{frame}
\frametitle{Reward hacking (suite)}

\begin{beamerboxesrounded}[scheme=madrid]{Problématique}
Collaboration $\rightarrow$ oscillation de la balle entre les deux
buts, sans jamais marquer.
\end{beamerboxesrounded}

\vspace{0.3cm}

\pause

\textbf{Correction :} récompense \emph{zero-sum} — le gain de l'un
devient la perte de l'autre \nocite{littmanMarkovGamesFramework1994}

% Grande importance à la visualisation de l'agent dans l'environnement pour détecter ce type de dérives

\end{frame}

\subsection{PPO vs SAC}
\begin{frame}
\frametitle{PPO vs SAC : une divergence marquée}

\begin{itemize}
    \item \textbf{PPO} : progresse à travers tout le curriculum
    \item \textbf{SAC} : stagne dès le Palier I, entraînement interrompu
\end{itemize}

\vspace{0.5cm}

\textbf{Deux facteurs explicatifs :}
\begin{enumerate}
    \item Limites d'implémentation de \texttt{rlgym-sac}
    \item Espace continu brut (8 dim.) contre espace discret curaté
    (90 actions) pour PPO
\end{enumerate}

\end{frame}

\subsection{Résultats}
\begin{frame}
\frametitle{Résultats de l'agent PPO}

\begin{itemize}
    \item \textbf{3 matchs gagnés, 0 perdus} contre Element (finaliste RLBot 2022)
    \item \textbf{3 matchs gagnés, 0 perdus} contre Necto (vainqueur RLBot 2022, budget
        supérieur)
    \item \textbf{6 victoires sur 9} contre des joueurs humains
        (Or à Champion) — 79 buts marqués, 51 encaissés
\end{itemize}

\nocite{RLBotChampionship2022}

\vspace{0.5cm}

\pause

\textbf{Comportements émergents non récompensés :}
\begin{itemize}
    \item Gestion du boost plus économe qu'un joueur humain
    \item Usage systématique et précoce du \emph{powerslide}
    \item Apprentissage du \textit{flick}
\end{itemize}

\end{frame}

\section{Limites et perspectives}

\subsection{Limites}
\begin{frame}
\frametitle{Limites}

\begin{itemize}
    \item \textbf{Phase I :} 5 seeds seulement $\rightarrow$
    écarts-types élevés, estimations bruitées
    \item \textbf{Phase II :} aucun tuning systématique des
    hyperparamètres (coût prohibitif à cette échelle)
    \item \textbf{PPO vs SAC :} comparaison non tranchée
\end{itemize}

\end{frame}

\subsection{Perspectives}
\begin{frame}
\frametitle{Perspectives}

\begin{enumerate}
    \item \textbf{Corriger l'implémentation SAC} pour trancher
    réellement la comparaison
    \item \textbf{Apprentissage par imitation} (Video PreTraining) à
    partir de replays humains \nocite{bakerVideoPreTrainingVPT2022}
    \item \textbf{Entraînement multi-agent} : 2v2/3v3, diversification
    des partenaires (\emph{league training})
\end{enumerate}

\end{frame}


\section{Conclusion}
\begin{frame}
\frametitle{Conclusion}

\begin{itemize}
    \item \textbf{Aucun algorithme universel} : dépend de l'espace
    d'actions, du budget de calcul et de l'environnement
    \item La rareté du signal \textbf{peut être surmontée} par
    reward shaping et curriculum learning, \textbf{à condition} de rester
    vigilant face au reward hacking
\end{itemize}

\vspace{0.5cm}

\pause

\centering \textit{Du hasard d'un agent qui ne marque jamais, à un
agent qui bat des joueurs de niveau Champion.}

\end{frame}
