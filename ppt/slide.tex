\subsection{Problématique}
\begin{frame}
\frametitle{Le problème du \emph{sparse reward}}

Ce phénomène porte un nom : le \textbf{sparse reward} \nocite{narvekarCurriculumLearningReinforcement2020}

\vspace{0.3cm}

\begin{itemize}
    \item Quand l'objectif final n'est atteint que rarement (voire jamais
    au début), l'agent ne reçoit presque aucun signal exploitable.
    \item Résultat : \textbf{absence totale d'apprentissage observable},
    même après des millions de pas d'entraînement.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Problématique}

\begin{beamerboxesrounded}[upper=block head,lower=block body]{Problématique}
    Comment mener à bien l'apprentissage par renforcement dans un
    environnement à récompense rare, de la sélection et du réglage
    d'algorithmes standards jusqu'à la conception d'un signal de
    récompense pour un environnement avancé ?
\end{beamerboxesrounded}

\vspace{0.5cm}

\textbf{Deux axes complémentaires :}

\begin{enumerate}
    \item \textbf{Sélection et optimisation des modèles} : quel
    algorithme choisir (DQN, PPO, SAC) selon l'environnement, et
    comment régler ses hyperparamètres ?
    \item \textbf{Construction du signal de récompense} : comment
    densifier un signal rare sans dénaturer la tâche — et quel risque
    cela introduit ?
\end{enumerate}

\end{frame}

\begin{frame}
\frametitle{Démarche en deux phases}

\begin{enumerate}
    \item \textbf{Phase I} — Validation méthodologique sur
    un environnement plus simple \emph{LunarLander} : comparaison DQN / PPO / SAC, optimisation
    Optuna, robustesse multi-seeds.
    \item \textbf{Phase II} — Retour à \textbf{Rocket League} :
    reward shaping et curriculum learning pour surmonter la rareté
    structurelle du signal.
\end{enumerate}

\end{frame}

\section{Phase II}

\subsection{L'environnement Rocket League}
\begin{frame}
\frametitle{L'environnement Rocket League}

\begin{itemize}
    \item Simulation via \textbf{RocketSim} + \textbf{RLGym} : physique fidèle,
    entraînement \emph{headless} (sans rendu graphique) \nocite{ZealanLRocketSim2022,RLGymRLGym2020}
    \item Mode \textbf{1v1} : un agent affronte une copie de lui-même
    (\emph{self-play})
    \item Espace d'actions réduit par discrétisation :
    $1944 \rightarrow \textbf{90}$ actions pertinentes
    \item Fréquence de décision ramenée de 120~Hz à \textbf{15~Hz}
    (\emph{action repeat})
\end{itemize}

\vspace{0.5cm}

\begin{beamerboxesrounded}[upper=block head,lower=block body]{Rappel}
    Le seul signal naturel — marquer un but — est \textbf{extrêmement rare} :
    aucun apprentissage observable en RL pur.
\end{beamerboxesrounded}

\end{frame}

\subsection{Reward shaping et curriculum}
\begin{frame}
\frametitle{Reward shaping et curriculum}

\begin{center}
\begin{tabular}{|l|l|}
\hline
\textbf{Palier} & \textbf{Objectif} \\
\hline
I & Contrôle de base et contact avec la balle \\
II & Apprentissage du score (introduction du but) \\
III & Perfectionnement mécanique (boost, aérien, kickoff) \\
\hline
\end{tabular}
\end{center}

\vspace{0.5cm}

\begin{itemize}
    \item \textbf{Reward shaping} : densifier le signal rare avec des
    signaux intermédiaires (contact, vitesse vers la balle...)
    \item \textbf{Curriculum learning} : introduire ces signaux
    progressivement, sans dégrader les acquis des paliers précédents
\end{itemize}

\vspace{0.3cm}

\centering\footnotesize{Une introduction simultanée de tous les signaux
converge plus lentement.}

\end{frame}

\begin{frame}
\frametitle{Reward hacking (suite)}
\vspace{0.3cm}

\begin{beamerboxesrounded}[upper=block head,lower=block body]{Comportement observé}
    Les deux agents collaborent pour faire osciller la balle entre les
    deux buts, maximisant leur récompense individuelle sans jamais
    chercher à marquer.
\end{beamerboxesrounded}

\vspace{0.3cm}

\textbf{Correction :} reformulation en récompense \emph{zero-sum} —
le gain de l'un devient nécessairement la perte de l'autre \nocite{littmanMarkovGamesFramework1994}

\end{frame}

\subsection{PPO vs SAC}
\begin{frame}
\frametitle{PPO vs SAC : une divergence marquée}

\begin{itemize}
    \item \textbf{PPO} : progresse de manière cohérente à travers tout
    le curriculum
    \item \textbf{SAC} : stagne dès le Palier I, entraînement interrompu
\end{itemize}

\vspace{0.5cm}

\textbf{Deux facteurs explicatifs :}
\begin{enumerate}
    \item Limites d'implémentation de \texttt{rlgym-sac} (pas de
    gradient clipping, cible d'entropie fixe)
    \item Espace d'actions \textbf{continu brut} (8 dimensions) contre
    l'espace \textbf{discret curaté} (90 actions) de PPO
\end{enumerate}

\vspace{0.5cm}

\begin{beamerboxesrounded}[upper=block head,lower=block body]{Conclusion}
    Résultat non tranché : impossible de conclure à une infériorité
    intrinsèque de SAC sur cet environnement.
\end{beamerboxesrounded}

\end{frame}

\subsection{Résultats}
\begin{frame}
\frametitle{Résultats de l'agent PPO}


\begin{itemize}
    \item Victoires \textbf{3-0} contre Element (finaliste RLBot 2022)
    \item Victoires \textbf{3-0} contre Necto (vainqueur RLBot 2022,
    entraîné sur 3 modes de jeu et un budget supérieur)
    \item \textbf{6 victoires sur 9} contre des joueurs humains
    (Or à Champion)
\end{itemize}

\nocite{RLBotChampionship2022}

\vspace{0.5cm}

\textbf{Comportements émergents non récompensés explicitement :}
\begin{itemize}
    \item Gestion du boost plus économe qu'un joueur humain
    \item Utilisation systématique et précoce du \emph{powerslide}
\end{itemize}

\end{frame}

\section{Limites et perspectives}

\subsection{Limites}
\begin{frame}
\frametitle{Limites}

\begin{itemize}
    \item \textbf{Phase I :} seulement \textbf{5 seeds} $\rightarrow$
    écarts-types élevés (45 à 52 points), estimations encore bruitées
    \item \textbf{Phase II :} aucun tuning systématique des
    hyperparamètres (coût computationnel prohibitif à cette échelle + complexité avec le curriculum)
    \item \textbf{Comparaison PPO/SAC sur Rocket League :} non
    tranchée — limites d'implémentation vs limites intrinsèques de
    l'algorithme
\end{itemize}

\vspace{0.5cm}

\begin{beamerboxesrounded}[upper=block head,lower=block body]{À retenir}
    Un score d'optimisation ou un résultat isolé ne suffit jamais à
    conclure seul — la validation multi-seeds et la prudence
    d'interprétation restent essentielles.
\end{beamerboxesrounded}

\end{frame}

\subsection{Perspectives}
\begin{frame}
\frametitle{Perspectives}

\begin{enumerate}
    \item \textbf{Corriger l'implémentation SAC} : gradient clipping,
    standardisation effective des retours $\rightarrow$ trancher
    réellement la comparaison avec PPO
    \item \textbf{Apprentissage par imitation} (Video PreTraining) :
    initialiser l'agent à partir de replays humains plutôt que d'une
    politique aléatoire \nocite{bakerVideoPreTrainingVPT2022}
    \item \textbf{Entraînement multi-agent} : généraliser au 2v2/3v3
    et diversifier les partenaires d'entraînement (\emph{league
    training})
\end{enumerate}

\end{frame}


\section{Conclusion}
\begin{frame}
\frametitle{Conclusion}

\begin{center}
\emph{Explorer un domaine riche et vaste : de la sélection d'algorithmes
standards à la conception d'un signal de récompense pour un
environnement avancé.}
\end{center}

\vspace{0.5cm}

\begin{itemize}
    \item \textbf{Aucun algorithme universel} : le choix optimal dépend
    de l'espace d'actions, du budget de calcul et de la maturité de
    l'implémentation — bien plus que d'un score de tuning isolé
    \item La rareté du signal de récompense \textbf{peut être surmontée}
    par la combinaison du \emph{reward shaping} et du
    \emph{curriculum learning}...
    \item ...\textbf{à condition de rester vigilant} face au risque de
    \emph{reward hacking} qu'elle introduit
\end{itemize}

\vspace{0.5cm}

\begin{beamerboxesrounded}[upper=block head,lower=block body]{En résumé}
    De LunarLander à Rocket League : une démarche progressive et
    empirique, validée à chaque étape par l'observation directe des
    comportements de l'agent.
\end{beamerboxesrounded}

\end{frame}
